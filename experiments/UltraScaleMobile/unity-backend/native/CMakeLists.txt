cmake_minimum_required(VERSION 3.22.1)
project(ultrascale_unity_bridge LANGUAGES CXX)
add_library(ultrascale_unity_bridge SHARED ultrascale_unity_bridge.cpp)
target_compile_features(ultrascale_unity_bridge PRIVATE cxx_std_17)
target_compile_options(ultrascale_unity_bridge PRIVATE -Wall -Wextra -Werror)
set_target_properties(ultrascale_unity_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
